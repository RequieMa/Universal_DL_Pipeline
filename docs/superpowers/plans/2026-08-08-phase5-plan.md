"""Hyperparameter search — GridSearch, RandomSearch."""

from __future__ import annotations

import itertools
import logging
from abc import ABC, abstractmethod
from collections.abc import Iterator
from copy import deepcopy
from dataclasses import dataclass, field
from typing import TYPE_CHECKING, Any

import numpy as np

if TYPE_CHECKING:
    from pipeline.config import Config
    from pipeline.pipeline import BasePipeline

_logger = logging.getLogger(__name__)


@dataclass
class TrialResult:
    """Result of one hyperparameter trial.
    
    Attributes:
        trial_id: Zero-based trial index.
        config: The Config used for this trial.
        metrics: Metric name → value from the trained pipeline.
        score: The scoring metric value (higher is better).
    """
    trial_id: int
    config: Config
    metrics: dict[str, float]
    score: float


@dataclass
class SearchResult:
    """Results of a hyperparameter search.
    
    Attributes:
        trials: All trial results, sorted by score descending.
        best_trial: The trial with the highest score.
        best_config: The Config from the best trial.
        best_score: The best score achieved.
    """
    trials: list[TrialResult] = field(default_factory=list)
    best_trial: TrialResult | None = None
    best_config: Config | None = None
    best_score: float = float("-inf")
    
    def to_dataframe(self):
        """Return trials as a pandas DataFrame for analysis.
        
        Returns:
            DataFrame with columns: trial_id, score, and each metric.
        """
        import pandas as pd
        rows = []
        for t in self.trials:
            row = {"trial_id": t.trial_id, "score": t.score, **t.metrics}
            rows.append(row)
        return pd.DataFrame(rows)


class BaseSearch(ABC):
    """Abstract hyperparameter search.
    
    Subclasses implement :meth:`_generate_configs` to produce trial
    configs. :meth:`run` trains one pipeline per config and collects
    results.
    
    Args:
        pipeline_cls: A :class:`BasePipeline` subclass (not instance).
        base_config: Baseline :class:`Config` shared across all trials.
        scoring: Metric key to optimize (default ``"accuracy"``).
    """
    
    def __init__(
        self,
        pipeline_cls: type[BasePipeline],
        base_config: Config,
        scoring: str = "accuracy",
    ) -> None:
        self.pipeline_cls = pipeline_cls
        self.base_config = base_config
        self.scoring = scoring
    
    @abstractmethod
    def _generate_configs(self) -> Iterator[tuple[int, Config]]:
        """Yield ``(trial_id, config)`` pairs."""
        ...
    
    def run(self) -> SearchResult:
        """Run the search. Trains one pipeline per generated config.
        
        Trial failures are caught and logged; the trial is skipped
        (not added to results).
        
        Returns:
            :class:`SearchResult` sorted by score descending.
        """
        result = SearchResult()
        
        for trial_id, config in self._generate_configs():
            _logger.info("Trial %d: %r", trial_id, config)
            try:
                pipeline = self.pipeline_cls(config)
                state = pipeline.run("train")
                
                score = state.metrics.get(self.scoring, 0.0)
                trial = TrialResult(
                    trial_id=trial_id,
                    config=deepcopy(config),
                    metrics=dict(state.metrics),
                    score=float(score),
                )
                result.trials.append(trial)
                
                if score > result.best_score:
                    result.best_score = score
                    result.best_trial = trial
                    result.best_config = deepcopy(config)
            except Exception:
                _logger.exception("Trial %d failed, skipping", trial_id)
        
        result.trials.sort(key=lambda t: t.score, reverse=True)
        return result


class GridSearch(BaseSearch):
    """Exhaustive search over the Cartesian product of a parameter grid.
    
    Usage::
    
        grid = {"learning_rate": [0.1, 0.01], "batch_size": [16, 32]}
        search = GridSearch(MyPipeline, base_config, grid)
        result = search.run()
        print(result.best_config)
    
    Args:
        pipeline_cls: :class:`BasePipeline` subclass.
        base_config: Baseline configuration.
        param_grid: Dict mapping Config field names to lists of values.
        scoring: Metric key to optimize.
    """
    
    def __init__(
        self,
        pipeline_cls: type[BasePipeline],
        base_config: Config,
        param_grid: dict[str, list[Any]],
        scoring: str = "accuracy",
    ) -> None:
        super().__init__(pipeline_cls, base_config, scoring)
        if not param_grid:
            raise ValueError("param_grid must not be empty")
        self._param_grid = param_grid
    
    def _generate_configs(self) -> Iterator[tuple[int, Config]]:
        keys = list(self._param_grid.keys())
        values = list(self._param_grid.values())
        for trial_id, combo in enumerate(itertools.product(*values)):
            config = deepcopy(self.base_config)
            for key, val in zip(keys, combo):
                setattr(config, key, val)
            yield trial_id, config


class RandomSearch(BaseSearch):
    """Random hyperparameter search.
    
    Samples ``n_trials`` random combinations from the parameter grid
    without replacement.
    
    Usage::
    
        search = RandomSearch(MyPipeline, base_config, param_grid, n_trials=10)
    
    Args:
        pipeline_cls: :class:`BasePipeline` subclass.
        base_config: Baseline configuration.
        param_grid: Dict mapping Config field names to lists of values.
        n_trials: Number of random trials to run.
        scoring: Metric key to optimize.
        seed: Random seed for reproducibility.
    """
    
    def __init__(
        self,
        pipeline_cls: type[BasePipeline],
        base_config: Config,
        param_grid: dict[str, list[Any]],
        n_trials: int = 10,
        scoring: str = "accuracy",
        seed: int = 42,
    ) -> None:
        super().__init__(pipeline_cls, base_config, scoring)
        if not param_grid:
            raise ValueError("param_grid must not be empty")
        self._param_grid = param_grid
        self._n_trials = n_trials
        self._rng = np.random.default_rng(seed)
    
    def _generate_configs(self) -> Iterator[tuple[int, Config]]:
        keys = list(self._param_grid.keys())
        all_combos = list(itertools.product(*self._param_grid.values()))
        total = len(all_combos)
        n = min(self._n_trials, total)
        if n < self._n_trials:
            _logger.warning(
                "n_trials=%d exceeds total combinations=%d, clamping to %d",
                self._n_trials, total, n,
            )
        
        indices = self._rng.choice(total, size=n, replace=False)
        for trial_id, idx in enumerate(indices):
            combo = all_combos[idx]
            config = deepcopy(self.base_config)
            for key, val in zip(keys, combo):
                setattr(config, key, val)
            yield trial_id, config

**Exact implementation:** VotingEnsemble.__init__ takes `models: list[ModelProtocol]` + `mode: str = "hard"`. `predict(inputs)` runs all models, for hard mode returns mode of argmax predictions, for soft mode averages probabilities then argmax.
