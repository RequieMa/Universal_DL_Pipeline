### Task 2: serve.py — FastAPI OpenAI-compatible server

**Files:** `examples/m6-llm-distilgpt2/serve.py`

Standalone FastAPI script:
```python
"""Serve a fine-tuned distilgpt2 model via OpenAI-compatible API.

Usage:
    PYTHONPATH=. python serve.py
    # Then: curl http://localhost:8000/v1/chat/completions -H "Content-Type: application/json" -d '{"model":"distilgpt2","messages":[{"role":"user","content":"Hello"}]}'
"""

from fastapi import FastAPI
from pydantic import BaseModel
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer
from pipeline.adapters.torch_adapter import TorchModel
from pipeline.adapters.torch_adapter import TorchCheckpoint  # Phase 4 checkpoint

app = FastAPI(title="distilgpt2-instruct")

# Load model once at startup
@app.on_event("startup")
async def load_model():
    global model, tokenizer
    DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
    base = AutoModelForCausalLM.from_pretrained("distilgpt2").to(DEVICE)
    tokenizer = AutoTokenizer.from_pretrained("distilgpt2")
    tokenizer.pad_token = tokenizer.eos_token
    model = TorchModel(base)
    # Load fine-tuned checkpoint if available
    from pipeline.config import Config
    from pipeline.pipeline import PipelineState
    from pathlib import Path
    ckpt = Path("distilgpt2-instruct.pt")
    if ckpt.exists():
        state = PipelineState(config=Config())
        state.model = model
        TorchCheckpoint.load(state, ckpt)

class Message(BaseModel):
    role: str
    content: str

class ChatRequest(BaseModel):
    model: str = "distilgpt2"
    messages: list[Message]
    max_tokens: int = 128

class ChatResponse(BaseModel):
    choices: list[dict]

@app.post("/v1/chat/completions")
async def chat(req: ChatRequest):
    prompt = req.messages[-1].content if req.messages else ""
    inputs = tokenizer(prompt, return_tensors="pt").to(model._module.device)
    with torch.no_grad():
        outputs = model._module.generate(
            **inputs, max_new_tokens=req.max_tokens,
            do_sample=True, temperature=0.7, top_p=0.9,
            pad_token_id=tokenizer.eos_token_id,
        )
    response_text = tokenizer.decode(outputs[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)
    return ChatResponse(choices=[{"message": {"role": "assistant", "content": response_text}}])

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
